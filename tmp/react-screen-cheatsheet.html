067In-flight dedupe· same key while pending → same promise; forget on settle
export function dedupe<T>(fn: (key: string) => Promise<T>) {
  const inFlight = new Map<string, Promise<T>>();

  return (key: string): Promise<T> => {
    const existing = inFlight.get(key);
    if (existing) return existing;                 // SAME promise object, not a wrapper

    const p = fn(key).finally(() => inFlight.delete(key));   // forget on settle
    inFlight.set(key, p);                          // (also on rejection → no poison)
    return p;
  };
}

// dedupe ≠ cache: this forgets as soon as the request settles; a cache remembers
// finally() on reject still deletes → the next call retries instead of re-throwing
070Stale-while-revalidate· cached value now, refresh in background; first call waits
export function staleWhileRevalidate<T>(fetcher: (key: string) => Promise<T>) {
  const cache    = new Map<string, { value: T }>();   // wrapper → undefined is a value
  const inFlight = new Map<string, Promise<T>>();     // collapse concurrent refreshes

  const refresh = (key: string): Promise<T> => {
    const p = fetcher(key)
      .then(v => { cache.set(key, { value: v }); return v; })
      .finally(() => inFlight.delete(key));
    inFlight.set(key, p);
    return p;
  };

  return (key: string): Promise<T> => {
    const hit = cache.get(key);
    if (hit) {
      if (!inFlight.has(key)) refresh(key).catch(() => {});   // bg refresh, error swallowed
      return Promise.resolve(hit.value);                      // stale NOW
    }
    return inFlight.get(key) ?? refresh(key);                 // first call: wait
  };
}
077Retry with backoff· exponential · cap · jitter · retry predicate · injectable sleep
export async function retry<T>(fn: (attempt: number) => Promise<T>, o: RetryOptions) {
  const { retries, baseMs, maxMs = Infinity, jitter = false,
          isRetryable = () => true, sleep = defaultSleep, random = Math.random } = o;
  let lastError: unknown;

  for (let attempt = 0; attempt <= retries; attempt++) {     // total = retries + 1
    try {
      return await fn(attempt);
    } catch (error) {
      lastError = error;
      if (!isRetryable(error)) throw error;                  // fail fast
      if (attempt === retries) break;                        // no sleep after last try
      let delay = Math.min(baseMs * 2 ** attempt, maxMs);    // exponential, capped
      if (jitter) delay = Math.floor(delay * random());      // spread the herd
      await sleep(delay);                                    // injectable → fake timers
    }
  }
  throw lastError;
}
078Concurrency pool· ≤ k in flight · results in input order · PoolError after ALL settle
export function runPool<T>(tasks: Array<Task<T>>, k: number): Promise<T[]> {
  if (k < 1) throw new RangeError('k must be at least 1');    // sync throw by contract
  return run(tasks, k);
}
async function run<T>(tasks: Array<Task<T>>, k: number): Promise<T[]> {
  const results = new Array<T>(tasks.length);               // input order by index
  const errors: unknown[] = [];
  const executing = new Set<Promise<void>>();
  const all: Promise<void>[] = [];

  for (let i = 0; i < tasks.length; i++) {
    const p = Promise.resolve().then(() => tasks[i]())       // factories start lazily
      .then(r => { results[i] = r; }, e => { errors[i] = e; });
    all.push(p); executing.add(p); p.finally(() => executing.delete(p));
    if (executing.size >= k) await Promise.race(executing);  // slot refill on settle
  }
  await Promise.all(all);                                    // wait for everything
  if (errors.length) throw new PoolError(errors.filter(Boolean));
  return results;
}
076Cancelable promise· settle like inner promise OR AbortError on abort · cleanup exactly once
export function cancelable<T>(factory: () => Promise<T>, signal: AbortSignal,
                              cleanup?: () => void): Promise<T> {
  return new Promise<T>((resolve, reject) => {
    let settled = false;
    const settle = (fn: () => void) => {        // ONE gate for value / error / abort
      if (settled) return;                      // first caller wins, rest are no-ops
      settled = true;
      signal.removeEventListener('abort', onAbort);   // no leaked listeners
      cleanup?.();                              // exactly once
      fn();
    };
    const onAbort = () => settle(() => reject(new AbortError()));

    if (signal.aborted) { onAbort(); return; }  // already aborted: factory never runs
    signal.addEventListener('abort', onAbort, { once: true });

    factory().then(
      value => settle(() => resolve(value)),
      error => settle(() => reject(error)),
    );
  });
}
072Cancellable pipeline· sequential stages · abort between stages · cleanup in reverse
export async function runPipeline<R>(steps: PipelineStep[], input: unknown, signal: AbortSignal) {
  if (signal.aborted) throw new AbortError();
  const cleanups: Array<() => void | Promise<void>> = []; let off = () => {};
  const aborted = new Promise<never>((_, reject) => {        // raced against every step
    const onAbort = () => reject(new AbortError());
    signal.addEventListener('abort', onAbort, { once: true });
    off = () => signal.removeEventListener('abort', onAbort);
  });
  aborted.catch(() => {});                                    // no unhandled rejection
  try {
    let result: unknown = input;
    for (const step of steps) {
      if (signal.aborted) throw new AbortError();             // check BETWEEN stages
      const p = step.run(result, signal); p.catch(() => {});  // stage also gets the signal
      if (step.cleanup) cleanups.push(step.cleanup.bind(step));   // only if it started
      result = await Promise.race([p, aborted]);
    }
    return result as R;
  } finally {
    off();
    for (let i = cleanups.length - 1; i >= 0; i--) await cleanups[i]();   // reverse order
  }
}
066Timeout wrapper· race against a timer · TimeoutError · clear timer on settle
export class TimeoutError extends Error {
  constructor(msg = 'Operation timed out') { super(msg); this.name = 'TimeoutError'; }
}

export function withTimeout<T>(promise: Promise<T>, ms: number): Promise<T> {
  return new Promise((resolve, reject) => {
    const timer = setTimeout(() => reject(new TimeoutError()), ms);
    promise
      .then(v => { resolve(v); clearTimeout(timer); })   // clear → nothing leaks,
      .catch(e => { reject(e); clearTimeout(timer); });  //   tests/process can exit
  });
}
